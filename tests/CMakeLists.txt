    message(STATUS "Qt6 not found - tests are not built")
    return()
endif()

if(NOT Qt6Sql_FOUND)
    message(STATUS "Qt6 Sql not found - tests that use SQLite are disabled")
endif()

qt_standard_project_setup()

function(pb_add_test name source)
    add_executable(${name} ${source})
    if(WIN32)
        set_target_properties(${name} PROPERTIES WIN32_EXECUTABLE FALSE)
    endif()
    target_link_libraries(${name} PRIVATE pb_core Qt6::Test Qt6::Sql)
    add_test(NAME ${name} COMMAND ${name})
endfunction()
